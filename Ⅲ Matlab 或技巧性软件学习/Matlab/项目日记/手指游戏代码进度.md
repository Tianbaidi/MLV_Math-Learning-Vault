%% 1. 核心辅助函数

function [a, b, player] = get_state_from_index(idx, pairs, n_pairs)

pairs_squared = n_pairs * n_pairs;

if idx <= pairs_squared

player = 1;

state_in_player = idx;

else

player = 2;

state_in_player = idx - pairs_squared;

end

a_idx = ceil(state_in_player / n_pairs);

b_idx = mod(state_in_player - 1, n_pairs) + 1;

a = pairs(a_idx, :);

b = pairs(b_idx, :);

end

function result = iif(condition, true_val, false_val)

if condition

result = true_val;

else

result = false_val;

end

end

function [a_new, b_new, player_new, immediate_win] = execute_action(a, b, player, action)

a_new = a;

b_new = b;

player_new = 3 - player;

immediate_win = 0;

if player == 1

switch action

case 1

new_val = a(1) + b(1);

case 2

new_val = a(1) + b(2);

case 3

new_val = a(2) + b(1);

case 4

new_val = a(2) + b(2);

end

if new_val == 10

immediate_win = 1;

a_new = [];

b_new = [];

return;

elseif new_val > 10

new_val = new_val - 10;

end

if action == 1 || action == 2

a_new = sort([new_val, a(2)]);

else

a_new = sort([a(1), new_val]);

end

else

switch action

case 1

new_val = b(1) + a(1);

case 2

new_val = b(1) + a(2);

case 3

new_val = b(2) + a(1);

case 4

new_val = b(2) + a(2);

end

if new_val == 10

immediate_win = 2;

a_new = [];

b_new = [];

return;

elseif new_val > 10

new_val = new_val - 10;

end

if action == 1 || action == 2

b_new = sort([new_val, b(2)]);

else

b_new = sort([b(1), new_val]);

end

end

end

original_a = a;

original_player = player;

can_kill = false;

steps_to_kill = inf;

killer = 0; % 1:甲绝杀, 2:乙绝杀

% 使用带有深度限制的极小极大搜索

function [best_value, best_steps] = minimax_search(a_curr, b_curr, player_curr, depth, max_depth, alpha, beta)

% 极小极大搜索，返回当前状态对当前玩家的价值和达到这个价值的步数

% 检查是否达到深度限制

if depth >= max_depth

% 深度限制内没有获胜，返回一个中性值

best_value = 0; % 中性值，表示没有明显的优势

best_steps = inf;

return;

end

% 检查当前玩家是否能立即获胜

if player_curr == 1

for i = 1:2

for j = 1:2

if a_curr(i) + b_curr(j) == 10

best_value = 1; % 甲立即获胜

best_steps = depth;

return;

end

end

end

else

for i = 1:2

for j = 1:2

if b_curr(i) + a_curr(j) == 10

best_value = 0; % 乙立即获胜，甲输

best_steps = depth;

return;

end

end

end

end

if player_curr == 1

% 甲行动：最大化自己的获胜概率

best_value = -inf;

best_steps = inf;

for action = 1:4

[a_next, b_next, player_next, immediate_win] = execute_action(a_curr, b_curr, player_curr, action);

if immediate_win == 1

value = 1;

steps = depth + 1;

elseif immediate_win == 2

value = 0;

steps = depth + 1;

else

idx = state2idx(a_next(1), a_next(2), b_next(1), b_next(2), player_next);

[value_child, steps_child] = minimax_search(a_next, b_next, player_next, depth + 1, max_depth, alpha, beta);

value = value_child;

steps = steps_child;

end

if value > best_value || (value == best_value && steps < best_steps)

best_value = value;

best_steps = steps;

end

alpha = max(alpha, best_value);

if beta <= alpha

break; % α-β剪枝

end

end

else

% 乙行动：最小化甲的获胜概率

best_value = inf;

value = 0; % 乙立即获胜

steps = depth + 1;

elseif immediate_win == 1

value = 1; % 甲立即获胜

steps = depth + 1;

else

idx = state2idx(a_next(1), a_next(2), b_next(1), b_next(2), player_next);

[value_child, steps_child] = minimax_search(a_next, b_next, player_next, depth + 1, max_depth, alpha, beta);

value = value_child;

steps = steps_child;

end

if value < best_value || (value == best_value && steps < best_steps)

best_value = value;

best_steps = steps;

end

beta = min(beta, best_value);

if beta <= alpha

break; % α-β剪枝

end

end

end

end

end

killer = 2; % 乙绝杀

end

end

end

value = 0;

elseif immediate_win == 1

end

end
